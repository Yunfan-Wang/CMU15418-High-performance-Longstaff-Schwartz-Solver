cmake_minimum_required(VERSION 3.24)
project(LSMSolver VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LSM_ENABLE_CUDA "Build the global-regression CUDA solver" OFF)
option(LSM_ENABLE_MPI "Build the experimental iterative MPI solver" OFF)
include(CTest)

add_executable(lsm_cpu app/main.cpp)
target_include_directories(lsm_cpu PRIVATE app default/src)
if(MSVC)
  target_compile_options(lsm_cpu PRIVATE /O2 /W4)
  set_property(TARGET lsm_cpu PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(MINGW)
  target_compile_options(lsm_cpu PRIVATE -O3 -Wall -Wextra)
  target_link_options(lsm_cpu PRIVATE -static)
else()
  target_compile_options(lsm_cpu PRIVATE -O3 -Wall -Wextra)
endif()
install(TARGETS lsm_cpu RUNTIME DESTINATION .)

if(LSM_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(lsm_cuda app/main.cpp gpu/lsm.cu)
  target_include_directories(lsm_cuda PRIVATE app gpu)
  target_compile_definitions(lsm_cuda PRIVATE LSM_CUDA=1)
  set_target_properties(lsm_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON
                        CUDA_RUNTIME_LIBRARY Static)
  target_link_libraries(lsm_cuda PRIVATE CUDA::cudart_static)
  install(TARGETS lsm_cuda RUNTIME DESTINATION .)
endif()

if(LSM_ENABLE_MPI)
  find_package(MPI REQUIRED COMPONENTS CXX)
  add_executable(lsm_mpi advanced/src/main.cpp)
  target_include_directories(lsm_mpi PRIVATE advanced/src)
  target_link_libraries(lsm_mpi PRIVATE MPI::MPI_CXX)
endif()

if(BUILD_TESTING)
  add_test(NAME cpu_smoke COMMAND lsm_cpu --put --paths 10000 --steps 25 --json)
  add_test(NAME reject_zero_paths COMMAND lsm_cpu --paths 0)
  set_tests_properties(reject_zero_paths PROPERTIES WILL_FAIL TRUE)
  add_executable(lsm_numerics tests/numerics.cpp)
  target_include_directories(lsm_numerics PRIVATE default/src)
  add_test(NAME numerical_checks COMMAND lsm_numerics)
endif()
